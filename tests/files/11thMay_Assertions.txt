
//Assertion types - Hard assertions & soft assertions

//Hard assertion - If the assertion fails, the test will stop executing and be marked as failed.

//Soft Assertion - If the assertion fails, the test will continue executing and be marked as failed at the end of the test.

import { test, expect, Locator } from '@playwright/test'

test.skip('hard assertion example', async ({ page }) => {
  await page.goto('https://tutorialsninja.com/demo/')

 await expect(page).toHaveTitle(' Store') 
 //After this line, since assertion failed, rest the statements were not executed and test was marked as failed

 await expect(page).toHaveURL('https://tutorialsninja.com/demo/') 

 let logo: Locator=page.locator('//a[text()="Qafox.com"]')

 await expect(logo).toBeVisible() 

 await page.pause()
})

test('soft assertion example', async ({ page }) => {
  await page.goto('https://tutorialsninja.com/demo/')

 await expect.soft(page).toHaveTitle(' Store') 
 //After this line, since assertion failed, rest the statements will be executed and test was marked as failed

 await expect.soft(page).toHaveURL('https://tutorialsninja.com/demo/') 

 let logo: Locator=page.locator('//a[text()="Qafox.com"]')

 await expect.soft(logo).toBeVisible() 

 await page.pause()
})
